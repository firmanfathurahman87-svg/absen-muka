Integrasi Google Apps Script (Webhook)

Copy kode di bawah ke Script Editor di Google Sheets Anda agar sinkron secara otomatis!

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Auto-create Header if sheet is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID", "Tanggal", "NIP / NISN", "Nama", "Kategori", "Jabatan / Kelas", "Jam Masuk", "Jam Pulang", "Status"]);
    }
    
    var payload = data.payload || data;
    sheet.appendRow([
      payload.id || new Date().getTime(),
      payload.tanggal || "",
      payload.nip || "",
      payload.nama || "",
      payload.kategori || "",
      payload.jabatan || "",
      payload.jamMasuk || "-",
      payload.jamKeluar || "-",
      payload.status || "Hadir"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

Cara Pemasangan:

  1. Buka Google Sheets baru di Google Drive Anda.
  2. Klik menu Ekstensi > Apps Script.
  3. Hapus kode bawaan, lalu **paste** kode di atas. Klik ikon simpan (💾).
  4. Klik Terapkan (Deploy) > Web App Baru.
  5. Set Who has access (Siapa yang memiliki akses) menjadi Anyone (Siapa saja).
  6. Copy **URL Web App** dan paste ke Pengaturan Webhook aplikasi ini!
